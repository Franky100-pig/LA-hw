% 供 pandoc 的 -H 使用：让 LaTeX 能吃讲义里用到的少量 Unicode 符号。
% 默认正文字体（Latin Modern）没有 U+2713，缺字形会被吞成空白，
% 所以显式映射到 amssymb 的 \checkmark。
\usepackage{amssymb}
\usepackage{newunicodechar}
\newunicodechar{✓}{\ensuremath{\checkmark}}
\newunicodechar{✔}{\ensuremath{\checkmark}}
% Set 8 用到 ⇒（推出）与 ⟺，默认字体缺字形会吞成空白，映射到 amssymb
\newunicodechar{⇒}{\ensuremath{\Rightarrow}}
\newunicodechar{⟺}{\ensuremath{\iff}}
